Version Control

1.0; Grupo 3; 2025/04/19 ; First issue.

Code

L = [0.5 0.3];

thetalim = [-170 -170 -170 ;
             170  170  170];

current = [.5 .3 -45];

trelw = utoi([0.1 0.2 30.0]);
srelb = utoi([-0.1 0.3 0.0]);

trelg_1 = utoi([0.0  0.0 -90.0]);
trelg_2 = utoi([0.6  0.3  45.0]);
trelg_3 = utoi([-0.4 0.2   0.0]);
trelg_4 = utoi([0.8  1.4  30.0]);

srelg_1 = tmult(tinvert(srelb), tmult(trelw, trelg_1));
srelg_2 = tmult(tinvert(srelb), tmult(trelw, trelg_2));
srelg_3 = tmult(tinvert(srelb), tmult(trelw, trelg_3));
srelg_4 = tmult(tinvert(srelb), tmult(trelw, trelg_4));

[theta_1, ~, sol1] = solve_robot(srelg_1, current, trelw, srelb, L, thetalim);
[theta_2, ~, sol2] = solve_robot(srelg_2, current, trelw, srelb, L, thetalim);
[theta_3, ~, sol3] = solve_robot(srelg_3, current, trelw, srelb, L, thetalim);
[theta_4, ~, sol4] = solve_robot(srelg_4, current, trelw, srelb, L, thetalim);

if size(theta_1, 1) >= 3
    srelg_check1 = where_robot(theta_1, trelg_1, srelb, L)
end
if size(theta_2, 1) >= 3
    srelg_check2 = where_robot(theta_2, trelg_2, srelb, L)
end
if size(theta_3, 1) >= 3
    srelg_check3 = where_robot(theta_3, trelg_3, srelb, L)
end
if size(theta_4, 1) >= 3
    srelg_check4 = where_robot(theta_4, trelg_4, srelb, L)
end
