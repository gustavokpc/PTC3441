Script

r = 1/100;
Beff = [50 50 50];
Jeff = [14 14 14];
K = 2750;
N = 3;
freq = 100; % Hz
Ts = 1/freq;
wn = 2*pi*freq;
Kp = [(Jeff(1)*wn*wn)/K (Jeff(2)*wn*wn)/K (Jeff(3)*wn*wn)/K];
Kd = [(2*wn*Jeff(1) - Beff(1))/K (2*wn*Jeff(2) - Beff(2))/K (2*wn*Jeff(3) - Beff(3))/K];
F = 5;
T = 3;
L = [0.5; 0.3];
m = [4.6; 2.3; 1.0];

sim(['Ex5_LS3.slx']);
